// ============================================================
// Internationalization Logic
// ============================================================
function applyTranslations(lang) {
  if (!I18N[lang]) lang = "es";
  currentLang = lang;
  localStorage.setItem("lettingwork_lang", lang);
  const dict = I18N[lang];

  // Update Language Selector Button
  const langFlagEl = document.getElementById("current-lang-flag");
  const langTextEl = document.getElementById("current-lang-text");
  if (langFlagEl) langFlagEl.textContent = dict.lang_flag;
  if (langTextEl) langTextEl.textContent = dict.lang_name;

  // Highlight active option in dropdown
  document.querySelectorAll(".lang-option").forEach(btn => {
    const fnStr = btn.getAttribute("onclick") || "";
    if (fnStr.includes(`'${lang}'`)) {
      btn.classList.add("active");
    } else {
      btn.classList.remove("active");
    }
  });

  // Update all elements with data-i18n
  document.querySelectorAll("[data-i18n]").forEach(el => {
    const key = el.getAttribute("data-i18n");
    if (dict[key]) {
      el.innerHTML = dict[key];
    }
  });

  // Re-render recent list if present
  renderRecentUploads();
}

function toggleLangMenu() {
  const menu = document.getElementById("lang-dropdown");
  if (menu) {
    menu.classList.toggle("show");
  }
}

function selectLanguage(lang) {
  applyTranslations(lang);
  const menu = document.getElementById("lang-dropdown");
  if (menu) menu.classList.remove("show");
}

// Close language dropdown if clicked outside
document.addEventListener("click", (e) => {
  const wrapper = document.querySelector(".lang-dropdown-wrapper");
  if (wrapper && !wrapper.contains(e.target)) {
    const menu = document.getElementById("lang-dropdown");
    if (menu) menu.classList.remove("show");
  }
});

// Format bytes into human readable string
function formatBytes(bytes, decimals = 2) {
  if (!+bytes) return '0 Bytes';
  const k = 1024;
  const dm = decimals < 0 ? 0 : decimals;
  const sizes = ['Bytes', 'KB', 'MB', 'GB', 'TB'];
  const i = Math.floor(Math.log(bytes) / Math.log(k));
  return `${parseFloat((bytes / Math.pow(k, i)).toFixed(dm))} ${sizes[i]}`;
}

// Toast notification
function showToast(message, isError = false) {
  let toast = document.getElementById('toast');
  if (!toast) {
    toast = document.createElement('div');
    toast.id = 'toast';
    toast.className = 'toast';
    document.body.appendChild(toast);
  }
  toast.textContent = message;
  toast.style.borderColor = isError ? 'var(--accent-red)' : 'var(--accent-green)';
  toast.classList.add('show');
  setTimeout(() => {
    toast.classList.remove('show');
  }, 3200);
}

  if (backdrop) backdrop.classList.remove("active");
  document.body.style.overflow = "";
}

// User Authentication & Session
// ============================================================
function checkAuthStatus() {
  fetch("/api/auth/me")
    .then(res => res.json())
    .then(data => {
      if (data.authenticated && data.user) {
        currentUser = data.user;
        updateUserUI(currentUser);
      } else {
        currentUser = null;
        updateUserUI(null);
      }
    })
    .catch(() => {
      currentUser = null;
      updateUserUI(null);
    });
}

function updateUserUI(user) {
  const navUserLabel = document.getElementById("nav-user-label");
  const navAvatarChar = document.getElementById("nav-avatar-char");
  const drawerName = document.getElementById("drawer-user-name");
  const drawerSub = document.getElementById("drawer-user-sub");

  const drawerAuthActions = document.getElementById("drawer-auth-actions");
  const drawerLoggedActions = document.getElementById("drawer-logged-actions");
  const storageUsedText = document.getElementById("storage-used-text");
  const storageBarFill = document.getElementById("storage-bar-fill");

  const dict = I18N[currentLang] || I18N.es;

  if (user) {
    const initial = (user.username || "U")[0].toUpperCase();
    if (navUserLabel) navUserLabel.textContent = user.username;
    if (navAvatarChar) navAvatarChar.textContent = initial;

    if (drawerName) drawerName.textContent = user.username;
    if (drawerSub) drawerSub.textContent = user.email || "Usuario Verificado";
    if (drawerAvatar) {
      drawerAvatar.className = "drawer-avatar";
      drawerAvatar.textContent = initial;
    }

    if (drawerAuthActions) drawerAuthActions.style.display = "none";
    if (drawerLoggedActions) drawerLoggedActions.style.display = "block";

    // Update storage meter
    const usedBytes = user.storage_used_bytes || 0;

function fetchUserFiles() {
  if (!currentUser) return;
  fetch("/api/user/files")
    .then(res => res.json())
    .then(data => {
      if (data.success && Array.isArray(data.files)) {
        const localUploads = getStoredUploads();
        const combinedMap = new Map();
        
        data.files.forEach(f => {
          combinedMap.set(f.id, {
            id: f.id,
            original_name: f.original_name,
            size_bytes: f.size_bytes,
            share_url: `/f/${f.id}`,
            delete_token: f.delete_token,
            timestamp: f.upload_time
          });
        });

        localUploads.forEach(u => {
          if (!combinedMap.has(u.id)) {
            combinedMap.set(u.id, u);
          }
        });

        const updated = Array.from(combinedMap.values());
        localStorage.setItem(STORAGE_KEY, JSON.stringify(updated.slice(0, 40)));
        renderRecentUploads();
      }
    })
    .catch(() => {});
}

function getStoredUploads() {

function saveUploadRecord(record) {
  const uploads = getStoredUploads();
  const filtered = uploads.filter(u => u.id !== record.id);

  renderRecentUploads();
}

}

function escapeHtml(string) {
  const entityMap = {
    '&': '&amp;',
    '<': '&lt;',
    '>': '&gt;',
    '"': '&quot;',
    "'": '&#39;'
  };
  return String(string).replace(/[&<>"']/g, s => entityMap[s]);
}

function copyText(text) {
  const dict = I18N[currentLang] || I18N.es;
  if (navigator.clipboard) {
    navigator.clipboard.writeText(text).then(() => {
      showToast(dict.toast_copied);
    }).catch(() => {
      fallbackCopy(text);
    });
  } else {
    fallbackCopy(text);
  }
}

function fallbackCopy(text) {
  const dict = I18N[currentLang] || I18N.es;
  const textArea = document.createElement("textarea");
  textArea.value = text;
  document.body.appendChild(textArea);
  textArea.select();
  try {
    document.execCommand('copy');
    showToast(dict.toast_copied);
  } catch (err) {
    showToast('Failed to copy', true);
  }
  document.body.removeChild(textArea);
}

let currentXHR = null;

function setupUpload() {
  const dropzone = document.getElementById('dropzone');
  const fileInput = document.getElementById('file-input');
  if (!dropzone || !fileInput) return;

  dropzone.addEventListener('click', () => fileInput.click());

  ['dragenter', 'dragover'].forEach(eventName => {
    dropzone.addEventListener(eventName, (e) => {
      e.preventDefault();
      e.stopPropagation();
      dropzone.classList.add('dragover');
    }, false);
  });

  ['dragleave', 'drop'].forEach(eventName => {
    dropzone.addEventListener(eventName, (e) => {
      e.preventDefault();
      e.stopPropagation();
      dropzone.classList.remove('dragover');
    }, false);
  });

  dropzone.addEventListener('drop', (e) => {
    const files = e.dataTransfer.files;
    if (files.length > 0) {
      handleFileUpload(files[0]);
    }
  });

  fileInput.addEventListener('change', (e) => {
    if (e.target.files.length > 0) {
      handleFileUpload(e.target.files[0]);
    }
  });

  // Global drag-and-drop onto the window opens the upload modal automatically
  window.addEventListener('dragover', (e) => {
    e.preventDefault();
  }, false);

  window.addEventListener('drop', (e) => {
    // If drop occurred outside dropzone, still trigger upload
    if (!dropzone.contains(e.target)) {
      e.preventDefault();
      if (e.dataTransfer && e.dataTransfer.files.length > 0) {
        openUploadModal();
        handleFileUpload(e.dataTransfer.files[0]);
      }
    }
  }, false);
}

function handleFileUpload(file) {
  openUploadModal();

  const dropzone = document.getElementById('dropzone');
  const progressBox = document.getElementById('progress-box');
  const successCard = document.getElementById('success-card');
  const fileNamePreview = document.getElementById('upload-filename');
  const fileSizePreview = document.getElementById('upload-filesize');
  const progressBar = document.getElementById('progress-bar');
  const progressPercent = document.getElementById('progress-percent');
  const progressTransferred = document.getElementById('progress-transferred');

  if (dropzone) dropzone.style.display = 'none';
  if (successCard) successCard.style.display = 'none';
  if (progressBox) progressBox.style.display = 'block';

  fileNamePreview.textContent = file.name;
  fileSizePreview.textContent = formatBytes(file.size);
  progressBar.style.width = '0%';
  progressPercent.textContent = '0%';
  progressTransferred.textContent = `0 / ${formatBytes(file.size)}`;

  const startTime = Date.now();
  currentXHR = new XMLHttpRequest();

  currentXHR.upload.addEventListener('progress', (e) => {
    if (e.lengthComputable) {
      const percent = Math.round((e.loaded / e.total) * 100);
      progressBar.style.width = percent + '%';
      progressPercent.textContent = percent + '%';
      
      const elapsedSec = (Date.now() - startTime) / 1000;
      const speed = elapsedSec > 0 ? (e.loaded / elapsedSec) : 0;
      const ofLabel = currentLang === 'en' ? 'of' : (currentLang === 'es' ? 'de' : 'de');
      progressTransferred.textContent = `${formatBytes(e.loaded)} ${ofLabel} ${formatBytes(e.total)} (${formatBytes(speed)}/s)`;
    }
  });

  currentXHR.addEventListener('load', () => {
    if (currentXHR.status === 200) {
      try {
        const response = JSON.parse(currentXHR.responseText);
        if (response.success) {
          onUploadSuccess(response, file);
        } else {
          showToast(response.error || 'Upload error', true);
          resetUploadUI();
        }
      } catch (err) {
        showToast('Server response error', true);
        resetUploadUI();
      }
    } else {
      showToast('Error uploading file: ' + currentXHR.statusText, true);
      resetUploadUI();
    }
  });

  currentXHR.addEventListener('error', () => {
    showToast('Network error during upload', true);
    resetUploadUI();
  });

  currentXHR.open('POST', '/api/upload');
  currentXHR.setRequestHeader('X-Filename', encodeURIComponent(file.name));
  currentXHR.setRequestHeader('Content-Type', file.type || 'application/octet-stream');
  currentXHR.send(file);
}

function cancelUpload() {
  if (currentXHR) {
    currentXHR.abort();
    currentXHR = null;
    const dict = I18N[currentLang] || I18N.es;
    showToast(dict.btn_cancel);
    resetUploadUI();
  }
}

}

// Delete File API
function deleteFile(fileId, deleteToken) {
  const dict = I18N[currentLang] || I18N.es;
  if (!confirm(dict.confirm_delete)) {
    return;
  }

  fetch(`/api/file/${fileId}`, {
    method: 'DELETE',
    headers: {
      'X-Delete-Token': deleteToken
    }
  })
  .then(res => res.json())
  .then(data => {
    if (data.success) {
      showToast(dict.toast_deleted);
      removeStoredUpload(fileId);
      checkAuthStatus();
    } else {
      showToast(data.error || 'Failed to delete file', true);
    }
  })
  .catch(() => {
    showToast('Network error deleting file', true);
  });
}
