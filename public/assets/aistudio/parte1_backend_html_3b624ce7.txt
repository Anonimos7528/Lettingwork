import sqlite3
import os
import secrets
import hashlib
from typing import Optional, Dict, Any, List

DB_PATH = os.path.join(os.path.dirname(__file__), "lettingwork.db")

def get_connection() -> sqlite3.Connection:
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn

def hash_password(password: str, salt: Optional[str] = None) -> tuple[str, str]:
    if not salt:
        salt = secrets.token_hex(16)
    hashed = hashlib.sha256((salt + password).encode("utf-8")).hexdigest()
    return hashed, salt

def verify_password(password: str, stored_hash: str, salt: str) -> bool:
    hashed = hashlib.sha256((salt + password).encode("utf-8")).hexdigest()
    return secrets.compare_digest(hashed, stored_hash)

def init_db():
    with get_connection() as conn:
        conn.execute("""
            CREATE TABLE IF NOT EXISTS users (
                id TEXT PRIMARY KEY,
                username TEXT UNIQUE NOT NULL,
                email TEXT UNIQUE NOT NULL,
                password_hash TEXT NOT NULL,
                salt TEXT NOT NULL,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            );
        """)
        conn.execute("""
            CREATE TABLE IF NOT EXISTS sessions (
                token TEXT PRIMARY KEY,
                user_id TEXT NOT NULL,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            );
        """)
        conn.execute("""
            CREATE TABLE IF NOT EXISTS files (
                id TEXT PRIMARY KEY,
                original_name TEXT NOT NULL,
                stored_name TEXT NOT NULL,
                size_bytes INTEGER NOT NULL,
                mime_type TEXT,
                upload_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                download_count INTEGER DEFAULT 0,
                delete_token TEXT NOT NULL,
                user_id TEXT
            );
        """)
        # Safely ensure user_id column exists if table was created in older version
        try:
            conn.execute("ALTER TABLE files ADD COLUMN user_id TEXT;")
        except sqlite3.OperationalError:
            pass
        conn.commit()

def generate_id(prefix: str = "lw_", length: int = 8) -> str:
    return prefix + secrets.token_urlsafe(length)[:length]

# User Authentication functions
def create_user(username: str, email: str, password: str) -> Dict[str, Any]:
    username = username.strip()
    email = email.strip().lower()
    if len(username) < 3:
        raise ValueError("El nombre de usuario debe tener al menos 3 caracteres")
    if len(password) < 6:
        raise ValueError("La contraseña debe tener al menos 6 caracteres")
    if "@" not in email:
        raise ValueError("Correo electrónico inválido")

    hashed, salt = hash_password(password)
    user_id = generate_id("usr_", 10)

    with get_connection() as conn:
        try:
            conn.execute("""
                INSERT INTO users (id, username, email, password_hash, salt)
                VALUES (?, ?, ?, ?, ?)
            """, (user_id, username, email, hashed, salt))
            conn.commit()
        except sqlite3.IntegrityError:
            raise ValueError("El nombre de usuario o correo ya está registrado")

    return {"id": user_id, "username": username, "email": email}

def authenticate_user(identifier: str, password: str) -> Optional[Dict[str, Any]]:
    identifier = identifier.strip().lower()
    with get_connection() as conn:
        cursor = conn.execute("""
            SELECT id, username, email, password_hash, salt 
            FROM users 
            WHERE LOWER(username) = ? OR LOWER(email) = ?
        """, (identifier, identifier))
        row = cursor.fetchone()
        if not row:
            return None

        if verify_password(password, row["password_hash"], row["salt"]):
            return {"id": row["id"], "username": row["username"], "email": row["email"]}
    return None

def create_session(user_id: str) -> str:
    token = secrets.token_hex(32)
    with get_connection() as conn:
        conn.execute("INSERT INTO sessions (token, user_id) VALUES (?, ?)", (token, user_id))
        conn.commit()
    return token

def get_user_by_session(token: str) -> Optional[Dict[str, Any]]:
    if not token:
        return None
    with get_connection() as conn:
        cursor = conn.execute("""
            SELECT u.id, u.username, u.email, u.created_at
            FROM sessions s
            JOIN users u ON s.user_id = u.id
            WHERE s.token = ?
        """, (token,))
        row = cursor.fetchone()
        if row:
            return dict(row)
    return None

def delete_session(token: str) -> bool:
    with get_connection() as conn:
        conn.execute("DELETE FROM sessions WHERE token = ?", (token,))
        conn.commit()
        return True

def create_or_get_google_user(email: str, name: str) -> Dict[str, Any]:
    email = email.strip().lower()
    with get_connection() as conn:
        cursor = conn.execute("SELECT id, username, email FROM users WHERE LOWER(email) = ?", (email,))
        row = cursor.fetchone()
        if row:
            return dict(row)
        
        # New user from Google
        base_username = name.strip().replace(" ", "_").lower() if name else email.split("@")[0]
        clean_user = "".join(c for c in base_username if c.isalnum() or c in "_-")[:15] or "google_user"
        username = clean_user
        idx = 1
        while True:
            cursor = conn.execute("SELECT id FROM users WHERE LOWER(username) = ?", (username.lower(),))
            if not cursor.fetchone():
                break
            username = f"{clean_user}_{idx}"
            idx += 1

        user_id = generate_id("usr_", 10)
        dummy_hash, dummy_salt = hash_password(secrets.token_hex(20))
        conn.execute("""
            INSERT INTO users (id, username, email, password_hash, salt)
            VALUES (?, ?, ?, ?, ?)
        """, (user_id, username, email, dummy_hash, dummy_salt))
        conn.commit()
        return {"id": user_id, "username": username, "email": email}

# File operations
def save_file_metadata(original_name: str, stored_name: str, size_bytes: int, mime_type: str, user_id: Optional[str] = None) -> Dict[str, Any]:
    file_id = generate_id()
    delete_token = secrets.token_hex(16)
    
    with get_connection() as conn:
        conn.execute("""
            INSERT INTO files (id, original_name, stored_name, size_bytes, mime_type, delete_token, user_id)
            VALUES (?, ?, ?, ?, ?, ?, ?)
        """, (file_id, original_name, stored_name, size_bytes, mime_type, delete_token, user_id))
        conn.commit()
        
    return {
        "id": file_id,
        "original_name": original_name,
        "size_bytes": size_bytes,
        "mime_type": mime_type,
        "delete_token": delete_token,
        "user_id": user_id
    }

def get_file_by_id(file_id: str) -> Optional[Dict[str, Any]]:
    with get_connection() as conn:
        cursor = conn.execute("SELECT * FROM files WHERE id = ?", (file_id,))
        row = cursor.fetchone()
        if row:
            return dict(row)
    return None

def increment_download_count(file_id: str):
    with get_connection() as conn:
        conn.execute("UPDATE files SET download_count = download_count + 1 WHERE id = ?", (file_id,))
        conn.commit()

def delete_file_by_id(file_id: str, delete_token: str) -> Optional[str]:
    with get_connection() as conn:
        cursor = conn.execute("SELECT stored_name FROM files WHERE id = ? AND delete_token = ?", (file_id, delete_token))
        row = cursor.fetchone()
        if not row:
            return None
        
        stored_name = row["stored_name"]
        conn.execute("DELETE FROM files WHERE id = ?", (file_id,))
        conn.commit()
        return stored_name

def get_recent_files(limit: int = 10) -> List[Dict[str, Any]]:
    with get_connection() as conn:
        cursor = conn.execute("""
            SELECT id, original_name, size_bytes, mime_type, upload_time, download_count 
            FROM files 
            ORDER BY upload_time DESC 
            LIMIT ?
        """, (limit,))
        return [dict(row) for row in cursor.fetchall()]

def get_user_files(user_id: str) -> List[Dict[str, Any]]:
    with get_connection() as conn:
        cursor = conn.execute("""
            SELECT id, original_name, size_bytes, mime_type, upload_time, download_count, delete_token
            FROM files 
            WHERE user_id = ?
            ORDER BY upload_time DESC
        """, (user_id,))
        return [dict(row) for row in cursor.fetchall()]

def get_user_storage_usage(user_id: str) -> int:
    with get_connection() as conn:
        cursor = conn.execute("SELECT COALESCE(SUM(size_bytes), 0) as total FROM files WHERE user_id = ?", (user_id,))
        row = cursor.fetchone()
        return row["total"] if row else 0

import http.server
import socketserver
import os
import urllib.parse
import json
import uuid
import mimetypes
import email
import email.policy
from database import (

    get_file_by_id,
    increment_download_count,
    delete_file_by_id,
    get_recent_files,
    create_user,
    authenticate_user,
    create_session,
    get_user_by_session,
    delete_session,
    get_user_files,
    get_user_storage_usage,

)

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
UPLOADS_DIR = os.path.join(BASE_DIR, "uploads")
STATIC_DIR = os.path.join(BASE_DIR, "static")
TEMPLATES_DIR = os.path.join(BASE_DIR, "templates")
PORT = int(os.environ.get("PORT", 8080))

os.makedirs(UPLOADS_DIR, exist_ok=True)
os.makedirs(STATIC_DIR, exist_ok=True)
os.makedirs(TEMPLATES_DIR, exist_ok=True)

class ThreadedHTTPServer(socketserver.ThreadingMixIn, http.server.HTTPServer):
    daemon_threads = True
    allow_reuse_address = True

class LettingWorkHandler(http.server.BaseHTTPRequestHandler):
    protocol_version = "HTTP/1.1"

    def get_session_token(self) -> str | None:
        # 1. Check Authorization header
        auth_header = self.headers.get("Authorization", "")
        if auth_header.startswith("Bearer "):
            return auth_header[7:].strip()

        # 2. Check X-Auth-Token header
        x_token = self.headers.get("X-Auth-Token")
        if x_token:
            return x_token.strip()

        # 3. Check Cookies
        cookie_header = self.headers.get("Cookie", "")
        if cookie_header:
            cookies = cookie_header.split(";")
            for c in cookies:
                c = c.strip()
                if c.startswith("lw_session="):
                    return c[11:].strip()
        return None

    def get_current_user(self):
        token = self.get_session_token()
        if token:
            return get_user_by_session(token)
        return None

    def send_json(self, status_code: int, data: dict, set_cookie: str = None):
        body = json.dumps(data).encode("utf-8")
        self.send_response(status_code)
        self.send_header("Content-Type", "application/json; charset=utf-8")
        self.send_header("Content-Length", str(len(body)))
        self.send_header("Access-Control-Allow-Origin", "*")
        if set_cookie:
            self.send_header("Set-Cookie", set_cookie)
        self.end_headers()
        self.wfile.write(body)

    def send_file_response(self, file_path: str, content_type: str = "application/octet-stream", download_name: str = None):
        if not os.path.isfile(file_path):
            self.send_error(404, "File not found on disk")
            return

        file_size = os.path.getsize(file_path)
        self.send_response(200)
        self.send_header("Content-Type", content_type)
        self.send_header("Content-Length", str(file_size))
        self.send_header("Accept-Ranges", "bytes")
        
        if download_name:
            # RFC 5987 / 6266 encoding for international character sets
            encoded_name = urllib.parse.quote(download_name.encode('utf-8'))
            ascii_name = download_name.encode('ascii', 'replace').decode('ascii').replace('"', '')
            self.send_header("Content-Disposition", f'attachment; filename="{ascii_name}"; filename*=UTF-8\'\'{encoded_name}')
        
        self.end_headers()

        with open(file_path, "rb") as f:
            while chunk := f.read(64 * 1024):
                try:
                    self.wfile.write(chunk)
                except (BrokenPipeError, ConnectionResetError):
                    break

    def do_OPTIONS(self):
        self.send_response(204)
        self.send_header("Access-Control-Allow-Origin", "*")
        self.send_header("Access-Control-Allow-Methods", "GET, POST, DELETE, OPTIONS")
        self.send_header("Access-Control-Allow-Headers", "Content-Type, X-Filename, X-Delete-Token, Authorization, X-Auth-Token")
        self.end_headers()

    def do_GET(self):
        parsed_url = urllib.parse.urlparse(self.path)
        path = parsed_url.path

        # 1. Home page
        if path == "/" or path == "/index.html":
            index_path = os.path.join(TEMPLATES_DIR, "index.html")
            self.send_file_response(index_path, "text/html; charset=utf-8")
            return

        # 2. File share/download landing page: /f/<file_id>
        if path.startswith("/f/"):
            download_template = os.path.join(TEMPLATES_DIR, "download.html")
            self.send_file_response(download_template, "text/html; charset=utf-8")
            return

        # 3. Auth API: Get current user session
        if path == "/api/auth/me":
            user = self.get_current_user()
            if user:
                storage_used = get_user_storage_usage(user["id"])
                self.send_json(200, {
                    "authenticated": True,
                    "user": {
                        "id": user["id"],
                        "username": user["username"],
                        "email": user["email"],
                        "created_at": user["created_at"],
                        "storage_used_bytes": storage_used
                    }
                })
            else:
                self.send_json(200, {"authenticated": False})
            return

        # 4. User Files API: List user's private files
        if path == "/api/user/files":
            user = self.get_current_user()
            if not user:
                self.send_json(401, {"success": False, "error": "No autenticado"})
                return
            files = get_user_files(user["id"])
            storage_used = get_user_storage_usage(user["id"])
            self.send_json(200, {
                "success": True,
                "files": files,
                "storage_used_bytes": storage_used
            })
            return

        # 5. API: Get file metadata
        if path.startswith("/api/info/"):
            file_id = path[10:].strip("/")
            file_info = get_file_by_id(file_id)
            if not file_info:
                self.send_json(404, {"success": False, "error": "El archivo no existe o ha sido eliminado"})
                return
            
            safe_info = {
                "id": file_info["id"],
                "original_name": file_info["original_name"],
                "size_bytes": file_info["size_bytes"],
                "mime_type": file_info["mime_type"],
                "upload_time": file_info["upload_time"],
                "download_count": file_info["download_count"]
            }
            self.send_json(200, {"success": True, "file": safe_info})
            return

        # 6. API: Download binary file
        if path.startswith("/api/download/"):
            file_id = path[14:].strip("/")
            file_info = get_file_by_id(file_id)
            if not file_info:
                self.send_error(404, "Archivo no encontrado")
                return

            disk_path = os.path.join(UPLOADS_DIR, file_info["stored_name"])
            if not os.path.exists(disk_path):
                self.send_error(404, "El archivo físico no se encuentra en el servidor")
                return

            increment_download_count(file_id)
            mime_type = file_info["mime_type"] or "application/octet-stream"
            self.send_file_response(disk_path, mime_type, file_info["original_name"])
            return

        # 7. API: Public recent files (for guest preview)
        if path == "/api/recent":
            files = get_recent_files(10)
            self.send_json(200, {"success": True, "files": files})
            return

        # 8. Static files
        if path.startswith("/static/"):
            clean_subpath = path[8:].lstrip("/")
            full_path = os.path.abspath(os.path.join(STATIC_DIR, clean_subpath))
            if full_path.startswith(STATIC_DIR) and os.path.isfile(full_path):
                mime, _ = mimetypes.guess_type(full_path)
                self.send_file_response(full_path, mime or "application/octet-stream")
                return
            else:
                self.send_error(404, "Recurso no encontrado")
                return

        self.send_error(404, "Ruta no encontrada")

    def do_POST(self):
        parsed_url = urllib.parse.urlparse(self.path)
        path = parsed_url.path

        # 1. Auth: Register
        if path == "/api/auth/register":
            content_length = int(self.headers.get("Content-Length", 0))
            if not content_length:
                self.send_json(400, {"success": False, "error": "Petición vacía"})
                return

            try:
                body = json.loads(self.rfile.read(content_length).decode("utf-8"))
                username = body.get("username", "").strip()
                email = body.get("email", "").strip()
                password = body.get("password", "").strip()

                user = create_user(username, email, password)
                token = create_session(user["id"])
                cookie = f"lw_session={token}; Path=/; HttpOnly; SameSite=Lax; Max-Age=2592000"

                self.send_json(200, {
                    "success": True,
                    "token": token,
                    "user": user
                }, set_cookie=cookie)
            except ValueError as ve:
                self.send_json(400, {"success": False, "error": str(ve)})
            except Exception as e:
                self.send_json(500, {"success": False, "error": "Error interno del servidor"})
            return

        # 2. Auth: Login
        if path == "/api/auth/login":
            content_length = int(self.headers.get("Content-Length", 0))
            if not content_length:
                self.send_json(400, {"success": False, "error": "Petición vacía"})
                return

            try:
                body = json.loads(self.rfile.read(content_length).decode("utf-8"))
                identifier = body.get("identifier", "").strip()
                password = body.get("password", "").strip()

                user = authenticate_user(identifier, password)
                if not user:
                    self.send_json(401, {"success": False, "error": "Credenciales inválidas. Revisa usuario/email o contraseña"})
                    return

                token = create_session(user["id"])
                cookie = f"lw_session={token}; Path=/; HttpOnly; SameSite=Lax; Max-Age=2592000"

                self.send_json(200, {
                    "success": True,
                    "token": token,
                    "user": user
                }, set_cookie=cookie)
            except Exception as e:
                self.send_json(500, {"success": False, "error": "Error al procesar inicio de sesión"})
            return

        # 2.5 Auth: Google Sign-In
        if path == "/api/auth/google":
            content_length = int(self.headers.get("Content-Length", 0))
            if not content_length:
                self.send_json(400, {"success": False, "error": "Petición vacía"})
                return

            try:
                body = json.loads(self.rfile.read(content_length).decode("utf-8"))
                email = body.get("email", "").strip()
                name = body.get("name", "").strip()

                if not email or "@" not in email:
                    self.send_json(400, {"success": False, "error": "Correo electrónico de Google requerido"})
                    return

                user = create_or_get_google_user(email, name)
                token = create_session(user["id"])
                cookie = f"lw_session={token}; Path=/; HttpOnly; SameSite=Lax; Max-Age=2592000"

                self.send_json(200, {
                    "success": True,
                    "token": token,
                    "user": user
                }, set_cookie=cookie)
            except Exception as e:
                self.send_json(500, {"success": False, "error": "Error en autenticación con Google"})
            return

        # 3. Auth: Logout
        if path == "/api/auth/logout":
            token = self.get_session_token()
            if token:
                delete_session(token)
            cookie = "lw_session=; Path=/; HttpOnly; SameSite=Lax; Max-Age=0"
            self.send_json(200, {"success": True}, set_cookie=cookie)
            return

        # 4. Upload file
        if path == "/api/upload":
            current_user = self.get_current_user()
            user_id = current_user["id"] if current_user else None

            content_type_header = self.headers.get("Content-Type", "")
            content_length_header = self.headers.get("Content-Length")
            
            if not content_length_header:
                self.send_json(400, {"success": False, "error": "Cabecera Content-Length requerida"})
                return
                
            content_length = int(content_length_header)
            
            # Direct binary stream mode (via X-Filename header)
            raw_filename_header = self.headers.get("X-Filename")
            if raw_filename_header:
                original_name = urllib.parse.unquote(raw_filename_header)
                if not original_name:
                    original_name = "archivo_descarga"
                
                original_name = os.path.basename(original_name)
                ext = os.path.splitext(original_name)[1]
                stored_name = f"{uuid.uuid4().hex}{ext}"
                dest_path = os.path.join(UPLOADS_DIR, stored_name)
                
                bytes_read = 0
                chunk_size = 64 * 1024
                
                with open(dest_path, "wb") as f:
                    while bytes_read < content_length:
                        to_read = min(chunk_size, content_length - bytes_read)
                        chunk = self.rfile.read(to_read)
                        if not chunk:
                            break
                        f.write(chunk)
                        bytes_read += len(chunk)
                        
                mime_type = mimetypes.guess_type(original_name)[0] or "application/octet-stream"
                file_meta = save_file_metadata(original_name, stored_name, bytes_read, mime_type, user_id=user_id)
                
                self.send_json(200, {
                    "success": True,
                    "id": file_meta["id"],
                    "original_name": file_meta["original_name"],
                    "size_bytes": file_meta["size_bytes"],
                    "share_url": f"/f/{file_meta['id']}",
                    "delete_token": file_meta["delete_token"]
                })
                return

            # Multipart/form-data mode
            if "multipart/form-data" in content_type_header:
                raw_body = self.rfile.read(content_length)
                msg_bytes = f"Content-Type: {content_type_header}\r\n\r\n".encode("utf-8") + raw_body
                msg = email.message_from_bytes(msg_bytes, policy=email.policy.default)
                
                uploaded_part = None
                for part in msg.iter_parts():
                    if part.get_filename():
                        uploaded_part = part
                        break

                if not uploaded_part:
                    self.send_json(400, {"success": False, "error": "No se encontró ningún archivo en la petición"})
                    return

                original_name = os.path.basename(uploaded_part.get_filename() or "archivo")
                ext = os.path.splitext(original_name)[1]
                stored_name = f"{uuid.uuid4().hex}{ext}"
                dest_path = os.path.join(UPLOADS_DIR, stored_name)
                
                file_data = uploaded_part.get_payload(decode=True)
                with open(dest_path, "wb") as f:
                    f.write(file_data)

                mime_type = uploaded_part.get_content_type() or mimetypes.guess_type(original_name)[0] or "application/octet-stream"
                file_meta = save_file_metadata(original_name, stored_name, len(file_data), mime_type, user_id=user_id)

                self.send_json(200, {
                    "success": True,
                    "id": file_meta["id"],
                    "original_name": file_meta["original_name"],
                    "size_bytes": file_meta["size_bytes"],
                    "share_url": f"/f/{file_meta['id']}",
                    "delete_token": file_meta["delete_token"]
                })
                return

            self.send_json(400, {"success": False, "error": "Formato de subida no soportado. Usa cabecera X-Filename o multipart/form-data"})
            return

        self.send_error(404, "Ruta no encontrada")

    def do_DELETE(self):
        parsed_url = urllib.parse.urlparse(self.path)
        path = parsed_url.path

        if path.startswith("/api/file/"):
            file_id = path[10:].strip("/")
            delete_token = self.headers.get("X-Delete-Token")
            
            if not delete_token:
                content_length = int(self.headers.get("Content-Length", 0))
                if content_length > 0:
                    try:
                        body_data = json.loads(self.rfile.read(content_length).decode("utf-8"))
                        delete_token = body_data.get("delete_token")
                    except Exception:
                        pass

            if not delete_token:
                self.send_json(400, {"success": False, "error": "Token de borrado requerido (X-Delete-Token)"})
                return

            stored_name = delete_file_by_id(file_id, delete_token)
            if not stored_name:
                self.send_json(403, {"success": False, "error": "Archivo no encontrado o token inválido"})
                return

            # Remove physical file
            disk_path = os.path.join(UPLOADS_DIR, stored_name)
            if os.path.exists(disk_path):
                try:
                    os.remove(disk_path)
                except OSError:
                    pass

            self.send_json(200, {"success": True, "message": "Archivo eliminado correctamente"})
            return

        self.send_error(404, "Ruta no encontrada")

def run(port: int = PORT):
    init_db()
    httpd = None
    active_port = port
    candidate_ports = [port] + [p for p in [8081, 8082, 8000, 3000, 5000] if p != port]
    for p in candidate_ports:
        try:
            server_address = ("", p)
            httpd = ThreadedHTTPServer(server_address, LettingWorkHandler)
            active_port = p
            break
        except OSError:
            continue

    if not httpd:
        raise RuntimeError("No se pudo enlazar a ningún puerto disponible.")

    print(f"==================================================")
    print(f" LettingWork Cloud Storage v0.1.0 BETA está activo!")
    print(f" Abre en tu navegador:")
    print(f"  👉 http://localhost:{active_port}")
    print(f"==================================================")
    try:
        httpd.serve_forever()
    except KeyboardInterrupt:
        print("\nServidor detenido.")
        httpd.server_close()

if __name__ == "__main__":
    run()
